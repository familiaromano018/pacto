## Task 2: Página /ajuda — Server Component

**Files:** `src/app/ajuda/page.tsx`

- [ ] **Step 1: criar page.tsx**

```tsx
import type { Metadata } from 'next'
import { faqContent } from '@/lib/faq/content'
import HelpClient from './HelpClient'

export const metadata: Metadata = {
  title: 'Ajuda — Pacto',
  description: 'Como usar o Pacto: gastos, divisão, autorização entre parceiros, fechar mês, exportar PDF e mais.',
}

function buildJsonLd() {
  return {
    '@context': 'https://schema.org',
    '@type': 'FAQPage',
    mainEntity: faqContent.flatMap(section =>
      section.items.map(item => ({
        '@type': 'Question',
        name: item.question,
        acceptedAnswer: {
          '@type': 'Answer',
          text: item.answer,
        },
      }))
    ),
  }
}

export default function AjudaPage() {
  return (
    <>
      <script
        type="application/ld+json"
        dangerouslySetInnerHTML={{ __html: JSON.stringify(buildJsonLd()) }}
      />
      <HelpClient sections={faqContent} />
    </>
  )
}
```

## Task 3: HelpClient — Client Component com busca e accordion

**Files:** `src/app/ajuda/HelpClient.tsx`

- [ ] **Step 1: estrutura básica**

```tsx
'use client'

import { useState, useMemo } from 'react'
import Link from 'next/link'
import { tokens } from '@/lib/tokens'
import type { FAQSection, FAQItem } from '@/lib/faq/content'

interface Props {
  sections: FAQSection[]
}

export default function HelpClient({ sections }: Props) {
  const [query, setQuery] = useState('')

  const filtered = useMemo(() => {
    if (!query.trim()) return sections
    const q = query.toLowerCase()
    return sections
      .map(s => ({
        ...s,
        items: s.items.filter(i =>
          i.question.toLowerCase().includes(q) ||
          i.answer.toLowerCase().includes(q)
        ),
      }))
      .filter(s => s.items.length > 0)
  }, [query, sections])

  const totalMatches = useMemo(
    () => filtered.reduce((acc, s) => acc + s.items.length, 0),
    [filtered]
  )

  return (
    <main style={{ /* ... */ }}>
      {/* Header */}
      <div style={{ /* ... */ }}>
        <Link href="/" style={{ /* ... */ }}>← Voltar</Link>
        <h1>Central de Ajuda</h1>
        <p>Tudo o que você precisa saber sobre o Pacto.</p>
        <input
          type="search"
          inputMode="search"
          placeholder="Buscar... (ex: fechar o mês)"
          value={query}
          onChange={e => setQuery(e.target.value)}
          aria-label="Buscar na ajuda"
          style={{ /* ... */ }}
        />
        {query && (
          <div>{totalMatches} {totalMatches === 1 ? 'resultado' : 'resultados'}</div>
        )}
      </div>

      {/* Empty state */}
      {filtered.length === 0 && (
        <div>
          <p>Nenhum resultado para "{query}".</p>
          <button onClick={() => setQuery('')}>Limpar busca</button>
        </div>
      )}

      {/* Sections */}
      {filtered.map(section => (
        <section key={section.id} id={section.id}>
          <h2>{section.emoji} {section.title}</h2>
          {section.items.map(item => (
            <details key={item.id} id={item.id}>
              <summary>{item.question}</summary>
              <p style={{ whiteSpace: 'pre-wrap' }}>{item.answer}</p>
            </details>
          ))}
        </section>
      ))}

      {/* Footer */}
      <footer>
        <p>Ainda tem dúvida? <a href="https://wa.me/SEUNUMERO" target="_blank" rel="noopener">Fala com a gente no WhatsApp →</a></p>
      </footer>
    </main>
  )
}
```

## Riscos e mitigações

| Risco | Mitigação |
|---|---|
| Conteúdo do FAQ envelhecer (ex: trial duration mudar) | Centralizar valores variáveis em constantes (ex: `TRIAL_DAYS`) e referenciar via template literal. Em V1, valores são estáticos no texto — atualizar manualmente. |
| Search travar com muito conteúdo | 56 perguntas × ~200 chars = ~11KB total. `useMemo` + `includes()` é rápido. Não é problema. |
| Bundle size aumenta | Conteúdo (~15KB) só carrega em `/ajuda`. App principal não é afetado. |
| Links pra WhatsApp/suporte ficam quebrados | Manter como TODO / placeholder até ter canal definido. Ou apontar pro e-mail do user (`turbomusicbr@gmail.com`?). |
